cmake_minimum_required(VERSION 3.16)
project(anns VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ANNS_DOCA "Build real DOCA adapters and host/soc applications" OFF)
option(ANNS_SANITIZERS "Enable address/undefined sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -ffp-contract=off)
if(ANNS_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(anns_core src/common/config.cpp src/index/index.cpp src/search/search_core.cpp)
target_include_directories(anns_core PUBLIC include)
target_link_libraries(anns_core PUBLIC PkgConfig::JSONC)
add_library(anns_protocol src/protocol/messages.cpp src/protocol/handoff.cpp)
target_link_libraries(anns_protocol PUBLIC anns_core)
add_library(anns_runtime src/data/prefetch.cpp src/transport/read_transport.cpp src/runtime/session.cpp src/runtime/soc_worker.cpp src/runtime/host_coordinator.cpp)
target_link_libraries(anns_runtime PUBLIC anns_protocol)
add_library(anns_sim_backend src/transport/simulated/transport.cpp)
target_link_libraries(anns_sim_backend PUBLIC anns_runtime)
add_library(anns_report src/common/report.cpp)
target_link_libraries(anns_report PUBLIC anns_sim_backend)
foreach(app baseline sim)
  add_executable(anns_${app} apps/${app}_main.cpp)
  target_link_libraries(anns_${app} PRIVATE anns_report)
endforeach()
if(ANNS_DOCA)
  include(cmake/Doca.cmake)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(anns_tests tests/test_contracts.cpp)
  target_link_libraries(anns_tests PRIVATE anns_sim_backend)
  add_test(NAME contracts COMMAND anns_tests)
endif()
